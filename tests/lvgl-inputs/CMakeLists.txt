cmake_minimum_required(VERSION 3.16)
project(lge_inputs_host C CXX ASM)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../lvgl/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE}" lvgl)
add_subdirectory("${GENERATED_SOURCE}/ui" generated)
add_executable(lge_inputs_host main.c)
target_link_libraries(lge_inputs_host PRIVATE lvgl_game)
target_compile_options(lvgl_game_ui PRIVATE -Wall -Wextra -Werror -pedantic)
target_compile_options(lge_inputs_host PRIVATE -Wall -Wextra -Werror -pedantic)
