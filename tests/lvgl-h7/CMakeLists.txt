cmake_minimum_required(VERSION 3.16)
project(lge_h7_host_test C CXX ASM)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE}" lvgl)
option(LGE_TEST_STYLE_CACHE "Test the H7 speed-build LVGL style cache" OFF)
if(LGE_TEST_STYLE_CACHE)
  target_compile_definitions(lvgl PUBLIC LGE_TEST_STYLE_CACHE=1)
endif()
add_subdirectory("${GENERATED_SOURCE}/ui" generated)
target_compile_definitions(lvgl_game_ui PUBLIC LGE_DESKTOP_PLAYTEST=1)
option(LGE_TEST_TILE_BATCHES "Render the optional terrain strips" ON)
if(NOT LGE_TEST_TILE_BATCHES)
  target_compile_definitions(lvgl_game_ui PUBLIC LGE_TILE_BATCHES=0)
endif()
add_executable(lge_h7_host_test main.c cache.c)
target_link_libraries(lge_h7_host_test PRIVATE lvgl_game)
target_compile_options(lvgl_game_ui PRIVATE -Wall -Wextra -Werror -pedantic)
target_compile_options(lge_h7_host_test PRIVATE -Wall -Wextra -Werror -pedantic)
option(LGE_TEST_SANITIZERS "Check portable renderer/cache undefined behavior" OFF)
if(LGE_TEST_SANITIZERS)
  target_compile_options(lvgl_game_ui PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
  target_compile_options(lge_h7_host_test PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
  target_link_options(lge_h7_host_test PRIVATE -fsanitize=undefined)
endif()
